# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(psd_core OBJECT psd_core.c)
target_include_directories(
  psd_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                  ${CMAKE_SOURCE_DIR}/native/inc/doppler/psd)
target_link_libraries(psd_core PUBLIC
    fft_core
    spectral_core
    acc_trace_core)

# The C unit test; `ctest` runs it.
add_executable(test_psd_core
               ${CMAKE_SOURCE_DIR}/native/tests/psd/test_psd_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/psd/test_psd_symbols.c)
target_link_libraries(test_psd_core
                      PRIVATE psd_core fft_core
    spectral_core
    acc_trace_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_psd_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_psd_core COMMAND test_psd_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_psd_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/psd/bench_psd_core.c)
target_link_libraries(bench_psd_core
                      PRIVATE psd_core fft_core
    spectral_core
    acc_trace_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_psd_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
