# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(wfm_synth_core OBJECT wfm_synth_core.c)
target_include_directories(
  wfm_synth_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                        ${CMAKE_SOURCE_DIR}/native/inc/doppler/wfm_synth)
target_link_libraries(wfm_synth_core PUBLIC
    lo_core
    awgn_core
    pn_core
    fir_core
    resamp_core
    wfm_dsp_core
    wfm_frame_core
    cvt_core
    gold_core)

# The C unit test; `ctest` runs it.
add_executable(test_wfm_synth_core
               ${CMAKE_SOURCE_DIR}/native/tests/wfm_synth/test_wfm_synth_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/wfm_synth/test_wfm_synth_symbols.c)
target_link_libraries(test_wfm_synth_core
                      PRIVATE wfm_synth_core lo_core
    awgn_core
    pn_core
    fir_core
    resamp_core
    wfm_dsp_core
    wfm_frame_core
    cvt_core
    gold_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_wfm_synth_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_wfm_synth_core COMMAND test_wfm_synth_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_wfm_synth_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/wfm_synth/bench_wfm_synth_core.c)
target_link_libraries(bench_wfm_synth_core
                      PRIVATE wfm_synth_core lo_core
    awgn_core
    pn_core
    fir_core
    resamp_core
    wfm_dsp_core
    wfm_frame_core
    cvt_core
    gold_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_wfm_synth_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
