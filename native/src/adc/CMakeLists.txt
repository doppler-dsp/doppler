# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(adc_core OBJECT adc_core.c)
target_include_directories(
  adc_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                  ${CMAKE_SOURCE_DIR}/native/inc/doppler/adc)

# The C unit test; `ctest` runs it.
add_executable(test_adc_core
               ${CMAKE_SOURCE_DIR}/native/tests/adc/test_adc_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/adc/test_adc_symbols.c)
target_link_libraries(test_adc_core
                      PRIVATE adc_core ${JM_MATH_LIBRARY})
target_include_directories(test_adc_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_adc_core COMMAND test_adc_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_adc_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/adc/bench_adc_core.c)
target_link_libraries(bench_adc_core
                      PRIVATE adc_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_adc_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

include(${CMAKE_CURRENT_LIST_DIR}/adc_extra.cmake OPTIONAL)
