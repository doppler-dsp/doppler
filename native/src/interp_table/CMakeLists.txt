# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(interp_table_core OBJECT interp_table_core.c)
target_include_directories(
  interp_table_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                           ${CMAKE_SOURCE_DIR}/native/inc/doppler/interp_table)

# The C unit test; `ctest` runs it.
add_executable(test_interp_table_core
               ${CMAKE_SOURCE_DIR}/native/tests/interp_table/test_interp_table_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/interp_table/test_interp_table_symbols.c)
target_link_libraries(test_interp_table_core
                      PRIVATE interp_table_core ${JM_MATH_LIBRARY})
target_include_directories(test_interp_table_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_interp_table_core COMMAND test_interp_table_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_interp_table_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/interp_table/bench_interp_table_core.c)
target_link_libraries(bench_interp_table_core
                      PRIVATE interp_table_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_interp_table_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

include(${CMAKE_CURRENT_LIST_DIR}/interp_table_extra.cmake OPTIONAL)
