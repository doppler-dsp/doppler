# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(frame_core OBJECT frame_core.c)
target_include_directories(
  frame_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                    ${CMAKE_SOURCE_DIR}/native/inc/doppler/frame)
target_link_libraries(frame_core PUBLIC
    wfm_frame_core
    ccsds_tm_core
    conv_core
    rs_core
    pn_core
    gold_core
    cvt_core)

# The C unit test; `ctest` runs it.
add_executable(test_frame_core
               ${CMAKE_SOURCE_DIR}/native/tests/frame/test_frame_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/frame/test_frame_symbols.c)
target_link_libraries(test_frame_core
                      PRIVATE frame_core wfm_frame_core
    ccsds_tm_core
    conv_core
    rs_core
    pn_core
    gold_core
    cvt_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_frame_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_frame_core COMMAND test_frame_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_frame_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/frame/bench_frame_core.c)
target_link_libraries(bench_frame_core
                      PRIVATE frame_core wfm_frame_core
    ccsds_tm_core
    conv_core
    rs_core
    pn_core
    gold_core
    cvt_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_frame_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
