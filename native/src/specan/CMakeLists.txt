# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(specan_core OBJECT specan_core.c)
target_include_directories(
  specan_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                     ${CMAKE_SOURCE_DIR}/native/inc/doppler/specan)
target_link_libraries(specan_core PUBLIC
    ddc_core
    lo_core
    RateConverter_core
    resamp_core
    hbdecim_core
    cic_core
    fir_core
    resample_core
    agc_core
    dp_tlm_core
    hbdecim_r2c_core
    psd_core
    fft_core
    spectral_core
    acc_trace_core)

# The C unit test; `ctest` runs it.
add_executable(test_specan_core
               ${CMAKE_SOURCE_DIR}/native/tests/specan/test_specan_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/specan/test_specan_symbols.c)
target_link_libraries(test_specan_core
                      PRIVATE specan_core ddc_core
    lo_core
    RateConverter_core
    resamp_core
    hbdecim_core
    cic_core
    fir_core
    resample_core
    agc_core
    dp_tlm_core
    hbdecim_r2c_core
    psd_core
    fft_core
    spectral_core
    acc_trace_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_specan_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_specan_core COMMAND test_specan_core)

  bench_specan_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
