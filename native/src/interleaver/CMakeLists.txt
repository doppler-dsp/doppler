# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(interleaver_core OBJECT interleaver_core.c)
target_include_directories(
  interleaver_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                          ${CMAKE_SOURCE_DIR}/native/inc/doppler/interleaver)

# The C unit test; `ctest` runs it.
add_executable(test_interleaver_core
               ${CMAKE_SOURCE_DIR}/native/tests/interleaver/test_interleaver_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/interleaver/test_interleaver_symbols.c)
target_link_libraries(test_interleaver_core
                      PRIVATE interleaver_core ${JM_MATH_LIBRARY})
target_include_directories(test_interleaver_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_interleaver_core COMMAND test_interleaver_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_interleaver_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/interleaver/bench_interleaver_core.c)
target_link_libraries(bench_interleaver_core
                      PRIVATE interleaver_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_interleaver_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

# regenerated by just-makeit.
include(${CMAKE_CURRENT_LIST_DIR}/interleaver_extra.cmake OPTIONAL)
