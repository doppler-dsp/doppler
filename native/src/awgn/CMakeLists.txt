# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(awgn_core OBJECT awgn_core.c)
target_include_directories(
  awgn_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                   ${CMAKE_SOURCE_DIR}/native/inc/doppler/awgn)

# The C unit test; `ctest` runs it.
add_executable(test_awgn_core
               ${CMAKE_SOURCE_DIR}/native/tests/awgn/test_awgn_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/awgn/test_awgn_symbols.c)
target_link_libraries(test_awgn_core
                      PRIVATE awgn_core ${JM_MATH_LIBRARY})
target_include_directories(test_awgn_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_awgn_core COMMAND test_awgn_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_awgn_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/awgn/bench_awgn_core.c)
target_link_libraries(bench_awgn_core
                      PRIVATE awgn_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_awgn_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

# file: it is included here and never overwritten. This file is
# regenerated by just-makeit.
include(${CMAKE_CURRENT_LIST_DIR}/awgn_extra.cmake OPTIONAL)
