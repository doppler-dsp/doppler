# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(lo_core OBJECT lo_core.c)
target_include_directories(
  lo_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                 ${CMAKE_SOURCE_DIR}/native/inc/doppler/lo)

# The C unit test; `ctest` runs it.
add_executable(test_lo_core
               ${CMAKE_SOURCE_DIR}/native/tests/lo/test_lo_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/lo/test_lo_symbols.c)
target_link_libraries(test_lo_core
                      PRIVATE lo_core ${JM_MATH_LIBRARY})
target_include_directories(test_lo_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_lo_core COMMAND test_lo_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_lo_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/lo/bench_lo_core.c)
target_link_libraries(bench_lo_core
                      PRIVATE lo_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_lo_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

# file: it is included here and never overwritten. This file is
# regenerated by just-makeit.
include(${CMAKE_CURRENT_LIST_DIR}/lo_extra.cmake OPTIONAL)
