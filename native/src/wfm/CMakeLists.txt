# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

add_library(wfm_core OBJECT wfm_core.c bpsk_map.c qpsk_map.c wfm_awgn_amplitude.c wfm_ebno_to_snr_db.c mls_poly.c crc16.c rrc_h.c rc_h.c field_bits.c rrc_taps.c _sweep_ends.c dsss_spread.c)
target_include_directories(wfm_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc)

    wfm_core
    pn_core
    wfm_synth_core
    gold_core
    frame_core
    lo_core
    awgn_core
    fir_core
    resamp_core
    wfm_frame_core
    cvt_core
    ccsds_tm_core
    conv_core
    rs_core
    wfm_dsp_core
    Python3::NumPy)
target_include_directories(wfm PRIVATE ${CMAKE_SOURCE_DIR}/native/inc)

    LIBRARY_OUTPUT_DIRECTORY "${PYTHON_PACKAGE_DIR}/wfm"

    COMMENT "Copy wfm extension module")
