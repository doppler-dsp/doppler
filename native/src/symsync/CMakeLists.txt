# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(symsync_core OBJECT symsync_core.c)
target_include_directories(
  symsync_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                      ${CMAKE_SOURCE_DIR}/native/inc/doppler/symsync)
target_link_libraries(symsync_core PUBLIC
    nco_core
    farrow_core
    loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core)

# The C unit test; `ctest` runs it.
add_executable(test_symsync_core
               ${CMAKE_SOURCE_DIR}/native/tests/symsync/test_symsync_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/symsync/test_symsync_symbols.c)
target_link_libraries(test_symsync_core
                      PRIVATE symsync_core nco_core
    farrow_core
    loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_symsync_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_symsync_core COMMAND test_symsync_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_symsync_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/symsync/bench_symsync_core.c)
target_link_libraries(bench_symsync_core
                      PRIVATE symsync_core nco_core
    farrow_core
    loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_symsync_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
