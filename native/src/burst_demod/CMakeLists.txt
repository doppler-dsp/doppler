# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(burst_demod_core OBJECT burst_demod_core.c)
target_include_directories(
  burst_demod_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                          ${CMAKE_SOURCE_DIR}/native/inc/doppler/burst_demod)
target_link_libraries(burst_demod_core PUBLIC
    ppe_core
    fft_core
    spectral_core
    wfm_frame_core
    cvt_core
    pn_core
    gold_core
    mpsk_core)

# The C unit test; `ctest` runs it.
add_executable(test_burst_demod_core
               ${CMAKE_SOURCE_DIR}/native/tests/burst_demod/test_burst_demod_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/burst_demod/test_burst_demod_symbols.c)
target_link_libraries(test_burst_demod_core
                      PRIVATE burst_demod_core ppe_core
    fft_core
    spectral_core
    wfm_frame_core
    cvt_core
    pn_core
    gold_core
    mpsk_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_burst_demod_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_burst_demod_core COMMAND test_burst_demod_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_burst_demod_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/burst_demod/bench_burst_demod_core.c)
target_link_libraries(bench_burst_demod_core
                      PRIVATE burst_demod_core ppe_core
    fft_core
    spectral_core
    wfm_frame_core
    cvt_core
    pn_core
    gold_core
    mpsk_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_burst_demod_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
