# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(dll_core OBJECT dll_core.c)
target_include_directories(
  dll_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                  ${CMAKE_SOURCE_DIR}/native/inc/doppler/dll)
target_link_libraries(dll_core PUBLIC
    loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core)

# The C unit test; `ctest` runs it.
add_executable(test_dll_core
               ${CMAKE_SOURCE_DIR}/native/tests/dll/test_dll_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/dll/test_dll_symbols.c)
target_link_libraries(test_dll_core
                      PRIVATE dll_core loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_dll_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_dll_core COMMAND test_dll_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_dll_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/dll/bench_dll_core.c)
target_link_libraries(bench_dll_core
                      PRIVATE dll_core loop_filter_core
    lockdet_core
    detection_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_dll_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

include(${CMAKE_CURRENT_LIST_DIR}/dll_extra.cmake OPTIONAL)
