# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(ddcr_core OBJECT ddcr_core.c)
target_include_directories(
  ddcr_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                   ${CMAKE_SOURCE_DIR}/native/inc/doppler/ddcr)
target_link_libraries(ddcr_core PUBLIC
    lo_core
    RateConverter_core
    resamp_core
    hbdecim_core
    cic_core
    fir_core
    resample_core
    agc_core
    dp_tlm_core
    hbdecim_r2c_core)

# The C unit test; `ctest` runs it.
add_executable(test_ddcr_core
               ${CMAKE_SOURCE_DIR}/native/tests/ddcr/test_ddcr_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/ddcr/test_ddcr_symbols.c)
target_link_libraries(test_ddcr_core
                      PRIVATE ddcr_core lo_core
    RateConverter_core
    resamp_core
    hbdecim_core
    cic_core
    fir_core
    resample_core
    agc_core
    dp_tlm_core
    hbdecim_r2c_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_ddcr_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_ddcr_core COMMAND test_ddcr_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_ddcr_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/ddcr/bench_ddcr_core.c)
target_link_libraries(bench_ddcr_core
                      PRIVATE ddcr_core lo_core
    RateConverter_core
    resamp_core
    hbdecim_core
    cic_core
    fir_core
    resample_core
    agc_core
    dp_tlm_core
    hbdecim_r2c_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_ddcr_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
