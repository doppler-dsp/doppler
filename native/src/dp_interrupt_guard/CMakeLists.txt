# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(dp_interrupt_guard_core OBJECT dp_interrupt_guard_core.c)
target_include_directories(
  dp_interrupt_guard_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                                 ${CMAKE_SOURCE_DIR}/native/inc/doppler/dp_interrupt_guard)
target_link_libraries(dp_interrupt_guard_core PUBLIC
    $<TARGET_OBJECTS:dp_interrupt_obj>)

# The C unit test; `ctest` runs it.
add_executable(test_dp_interrupt_guard_core
               ${CMAKE_SOURCE_DIR}/native/tests/dp_interrupt_guard/test_dp_interrupt_guard_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/dp_interrupt_guard/test_dp_interrupt_guard_symbols.c)
target_link_libraries(test_dp_interrupt_guard_core
                      PRIVATE dp_interrupt_guard_core $<TARGET_OBJECTS:dp_interrupt_obj>
    ${JM_MATH_LIBRARY})
target_include_directories(test_dp_interrupt_guard_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_dp_interrupt_guard_core COMMAND test_dp_interrupt_guard_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_dp_interrupt_guard_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/dp_interrupt_guard/bench_dp_interrupt_guard_core.c)
target_link_libraries(bench_dp_interrupt_guard_core
                      PRIVATE dp_interrupt_guard_core $<TARGET_OBJECTS:dp_interrupt_obj>
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_dp_interrupt_guard_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
