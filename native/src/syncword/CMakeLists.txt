# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(syncword_core OBJECT syncword_core.c)
target_include_directories(
  syncword_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                       ${CMAKE_SOURCE_DIR}/native/inc/doppler/syncword)
target_link_libraries(syncword_core PUBLIC
    detection_core)

# The C unit test; `ctest` runs it.
add_executable(test_syncword_core
               ${CMAKE_SOURCE_DIR}/native/tests/syncword/test_syncword_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/syncword/test_syncword_symbols.c)
target_link_libraries(test_syncword_core
                      PRIVATE syncword_core detection_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_syncword_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_syncword_core COMMAND test_syncword_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_syncword_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/syncword/bench_syncword_core.c)
target_link_libraries(bench_syncword_core
                      PRIVATE syncword_core detection_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_syncword_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
