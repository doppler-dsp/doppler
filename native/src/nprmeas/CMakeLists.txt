# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(nprmeas_core OBJECT nprmeas_core.c)
target_include_directories(
  nprmeas_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                      ${CMAKE_SOURCE_DIR}/native/inc/doppler/nprmeas)
target_link_libraries(nprmeas_core PUBLIC
    fft_core
    spectral_core
    psd_core
    acc_trace_core)

# The C unit test; `ctest` runs it.
add_executable(test_nprmeas_core
               ${CMAKE_SOURCE_DIR}/native/tests/nprmeas/test_nprmeas_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/nprmeas/test_nprmeas_symbols.c)
target_link_libraries(test_nprmeas_core
                      PRIVATE nprmeas_core fft_core
    spectral_core
    psd_core
    acc_trace_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_nprmeas_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_nprmeas_core COMMAND test_nprmeas_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_nprmeas_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/nprmeas/bench_nprmeas_core.c)
target_link_libraries(bench_nprmeas_core
                      PRIVATE nprmeas_core fft_core
    spectral_core
    psd_core
    acc_trace_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_nprmeas_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
