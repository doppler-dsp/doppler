# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(doppler_channel_core OBJECT doppler_channel_core.c)
target_include_directories(
  doppler_channel_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                              ${CMAKE_SOURCE_DIR}/native/inc/doppler/doppler_channel)
target_link_libraries(doppler_channel_core PUBLIC
    resamp_core)

# The C unit test; `ctest` runs it.
add_executable(test_doppler_channel_core
               ${CMAKE_SOURCE_DIR}/native/tests/doppler_channel/test_doppler_channel_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/doppler_channel/test_doppler_channel_symbols.c)
target_link_libraries(test_doppler_channel_core
                      PRIVATE doppler_channel_core resamp_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_doppler_channel_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_doppler_channel_core COMMAND test_doppler_channel_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_doppler_channel_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/doppler_channel/bench_doppler_channel_core.c)
target_link_libraries(bench_doppler_channel_core
                      PRIVATE doppler_channel_core resamp_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_doppler_channel_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
