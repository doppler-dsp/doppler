# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(carrier_acq_core OBJECT carrier_acq_core.c)
target_include_directories(
  carrier_acq_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                          ${CMAKE_SOURCE_DIR}/native/inc/doppler/carrier_acq)
target_link_libraries(carrier_acq_core PUBLIC
    psd_core
    fft_core
    spectral_core
    acc_trace_core
    detection_core
    correlator_core
    fft2d_core
    dp_tlm_core
    cvt_core)

# The C unit test; `ctest` runs it.
add_executable(test_carrier_acq_core
               ${CMAKE_SOURCE_DIR}/native/tests/carrier_acq/test_carrier_acq_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/carrier_acq/test_carrier_acq_symbols.c)
target_link_libraries(test_carrier_acq_core
                      PRIVATE carrier_acq_core psd_core
    fft_core
    spectral_core
    acc_trace_core
    detection_core
    correlator_core
    fft2d_core
    dp_tlm_core
    cvt_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_carrier_acq_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_carrier_acq_core COMMAND test_carrier_acq_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_carrier_acq_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/carrier_acq/bench_carrier_acq_core.c)
target_link_libraries(bench_carrier_acq_core
                      PRIVATE carrier_acq_core psd_core
    fft_core
    spectral_core
    acc_trace_core
    detection_core
    correlator_core
    fft2d_core
    dp_tlm_core
    cvt_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_carrier_acq_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
