# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(uq15_to_f32_core OBJECT uq15_to_f32_core.c)
target_include_directories(
  uq15_to_f32_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                          ${CMAKE_SOURCE_DIR}/native/inc/doppler/uq15_to_f32)

# The C unit test; `ctest` runs it.
add_executable(test_uq15_to_f32_core
               ${CMAKE_SOURCE_DIR}/native/tests/uq15_to_f32/test_uq15_to_f32_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/uq15_to_f32/test_uq15_to_f32_symbols.c)
target_link_libraries(test_uq15_to_f32_core
                      PRIVATE uq15_to_f32_core ${JM_MATH_LIBRARY})
target_include_directories(test_uq15_to_f32_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_uq15_to_f32_core COMMAND test_uq15_to_f32_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_uq15_to_f32_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/uq15_to_f32/bench_uq15_to_f32_core.c)
target_link_libraries(bench_uq15_to_f32_core
                      PRIVATE uq15_to_f32_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_uq15_to_f32_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
