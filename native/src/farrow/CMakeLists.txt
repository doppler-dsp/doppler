# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(farrow_core OBJECT farrow_core.c)
target_include_directories(
  farrow_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                     ${CMAKE_SOURCE_DIR}/native/inc/doppler/farrow)

# The C unit test; `ctest` runs it.
add_executable(test_farrow_core
               ${CMAKE_SOURCE_DIR}/native/tests/farrow/test_farrow_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/farrow/test_farrow_symbols.c)
target_link_libraries(test_farrow_core
                      PRIVATE farrow_core ${JM_MATH_LIBRARY})
target_include_directories(test_farrow_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_farrow_core COMMAND test_farrow_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_farrow_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/farrow/bench_farrow_core.c)
target_link_libraries(bench_farrow_core
                      PRIVATE farrow_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_farrow_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

# Your own CMake for this directory goes in farrow_extra.cmake beside this
# file: it is included here and never overwritten. This file is
# regenerated by just-makeit.
include(${CMAKE_CURRENT_LIST_DIR}/farrow_extra.cmake OPTIONAL)
