# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(costas_core OBJECT costas_core.c)
target_include_directories(
  costas_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                     ${CMAKE_SOURCE_DIR}/native/inc/doppler/costas)
target_link_libraries(costas_core PUBLIC
    lo_core
    lockdet_core
    loop_filter_core
    dp_tlm_core)

# The C unit test; `ctest` runs it.
add_executable(test_costas_core
               ${CMAKE_SOURCE_DIR}/native/tests/costas/test_costas_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/costas/test_costas_symbols.c)
target_link_libraries(test_costas_core
                      PRIVATE costas_core lo_core
    lockdet_core
    loop_filter_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_costas_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_costas_core COMMAND test_costas_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_costas_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/costas/bench_costas_core.c)
target_link_libraries(bench_costas_core
                      PRIVATE costas_core lo_core
    lockdet_core
    loop_filter_core
    dp_tlm_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_costas_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
