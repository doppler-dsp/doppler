# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(conv_enc_core OBJECT conv_enc_core.c)
target_include_directories(
  conv_enc_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                       ${CMAKE_SOURCE_DIR}/native/inc/doppler/conv_enc)
target_link_libraries(conv_enc_core PUBLIC
    conv_core)

# The C unit test; `ctest` runs it.
add_executable(test_conv_enc_core
               ${CMAKE_SOURCE_DIR}/native/tests/conv_enc/test_conv_enc_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/conv_enc/test_conv_enc_symbols.c)
target_link_libraries(test_conv_enc_core
                      PRIVATE conv_enc_core conv_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_conv_enc_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_conv_enc_core COMMAND test_conv_enc_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_conv_enc_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/conv_enc/bench_conv_enc_core.c)
target_link_libraries(bench_conv_enc_core
                      PRIVATE conv_enc_core conv_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_conv_enc_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)
