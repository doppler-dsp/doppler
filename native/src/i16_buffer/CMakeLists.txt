# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code is all in its header, so there is nothing to
# compile: this target only passes on the include directories.
add_library(i16_buffer_core INTERFACE)
target_include_directories(
  i16_buffer_core INTERFACE ${CMAKE_SOURCE_DIR}/native/inc
                            ${CMAKE_SOURCE_DIR}/native/inc/doppler/i16_buffer)
target_link_libraries(i16_buffer_core INTERFACE
    dp_interrupt_guard_core)

# The C unit test; `ctest` runs it.
add_executable(test_i16_buffer_core
               ${CMAKE_SOURCE_DIR}/native/tests/i16_buffer/test_i16_buffer_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/i16_buffer/test_i16_buffer_symbols.c)
target_link_libraries(test_i16_buffer_core
                      PRIVATE i16_buffer_core dp_interrupt_guard_core
    ${JM_MATH_LIBRARY})
target_include_directories(test_i16_buffer_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_i16_buffer_core COMMAND test_i16_buffer_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_i16_buffer_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/i16_buffer/bench_i16_buffer_core.c)
target_link_libraries(bench_i16_buffer_core
                      PRIVATE i16_buffer_core dp_interrupt_guard_core
    ${JM_MATH_LIBRARY})
target_include_directories(
  bench_i16_buffer_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

# Your own CMake for this directory goes in i16_buffer_extra.cmake beside this
# file: it is included here and never overwritten. This file is
# regenerated by just-makeit.
include(${CMAKE_CURRENT_LIST_DIR}/i16_buffer_extra.cmake OPTIONAL)
