# The math library (libm), found as a full path. A bare `m` could be mistaken
# for a target of the same name (a module called `m`). Empty where the math
# functions are part of the C library itself (Windows, macOS).
find_library(JM_MATH_LIBRARY m)
if(NOT JM_MATH_LIBRARY)
  set(JM_MATH_LIBRARY "")
endif()

# The component's C code, compiled once and linked into both its
# Python extension and the project's C library.
add_library(acc_q15_core OBJECT acc_q15_core.c)
target_include_directories(
  acc_q15_core PUBLIC ${CMAKE_SOURCE_DIR}/native/inc
                      ${CMAKE_SOURCE_DIR}/native/inc/doppler/acc_q15)

# The C unit test; `ctest` runs it.
add_executable(test_acc_q15_core
               ${CMAKE_SOURCE_DIR}/native/tests/acc_q15/test_acc_q15_core.c
               ${CMAKE_SOURCE_DIR}/native/tests/acc_q15/test_acc_q15_symbols.c)
target_link_libraries(test_acc_q15_core
                      PRIVATE acc_q15_core ${JM_MATH_LIBRARY})
target_include_directories(test_acc_q15_core
                           PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/tests)
add_test(NAME test_acc_q15_core COMMAND test_acc_q15_core)

# The C benchmark; built, but not run by ctest.
add_executable(
  bench_acc_q15_core
  ${CMAKE_SOURCE_DIR}/native/benchmarks/acc_q15/bench_acc_q15_core.c)
target_link_libraries(bench_acc_q15_core
                      PRIVATE acc_q15_core ${JM_MATH_LIBRARY})
target_include_directories(
  bench_acc_q15_core PRIVATE ${CMAKE_SOURCE_DIR}/native/inc
                                   ${CMAKE_SOURCE_DIR}/native/benchmarks)

include(${CMAKE_CURRENT_LIST_DIR}/acc_q15_extra.cmake OPTIONAL)
